extends SceneTree
## Native captures of the main menu at the real tested sizes and the stress states.
const CJK := "白雪・シラユキ・テスト・ネーム・ロング・名前・白雪・シラユキ"
func _initialize() -> void: call_deferred("run")
func settle(n := 14) -> void:
	for i in n: await process_frame
func run() -> void:
	var out := OS.get_cmdline_user_args()[0].trim_prefix("--out=")
	DirAccess.make_dir_recursive_absolute(out)
	var main = load("res://main.tscn").instantiate()
	root.add_child(main)
	await settle()
	main.profile.data.xp = 1340
	main.profile.data.level = main.Profile.level_for_xp(1340)
	main.profile.data.credits = 2750
	main.profile.data.matches = 9
	main.profile.data.wins = 5
	main.profile.data.kills = 61
	main.profile.data.deaths = 44
	main.open_economy("profile")
	await settle()
	var panel = main.economy_panel
	var cases := []
	for dims in [Vector2i(960, 720), Vector2i(1024, 768), Vector2i(1280, 720), Vector2i(1560, 720), Vector2i(1600, 720)]:
		for scale in [1.0, 1.3]:
			cases.append([dims, scale, "Operator", false, false])
	cases.append([Vector2i(960, 720), 1.3, CJK, true, false])
	cases.append([Vector2i(960, 720), 1.3, "Operator", false, true])
	cases.append([Vector2i(1280, 720), 0.85, "Operator", true, false])
	for config in cases:
		main.profile.data.settings.ui_text_scale = config[1]
		main.ui.theme = main.build_theme()
		root.content_scale_size = config[0]
		root.size = config[0]
		main.profile.data.display_name = config[2]
		panel.api.token = "fixture-render-only" if config[3] else ""
		panel.account = {"display_name": CJK if config[3] else "", "credits": 18450, "xp": 200, "level": 1}
		main.menu_labels = {}
		if config[4]:
			for id in ["deploy", "lan", "inventory", "cases", "market", "trades", "record", "settings", "policy", "build", "record_title", "xp_to", "credits"]:
				main.menu_labels[id] = main.MENU_LABELS[id] + "   " + main.MENU_LABELS[id].substr(0, int(main.MENU_LABELS[id].length() * 0.6))
		main.show_menu()
		await settle()
		await RenderingServer.frame_post_draw
		var image := root.get_texture().get_image()
		var tag := "%dx%d-%d%s%s%s" % [config[0].x, config[0].y, roundi(config[1] * 100), "-account-cjk" if config[3] and config[2] == "Operator" == false else ("-account" if config[3] else ""), "-cjkname" if config[2] != "Operator" else "", "-pseudo" if config[4] else ""]
		image.save_png(out.path_join("menu-%s.png" % tag))
		print("CAPTURE menu-%s %dx%d side_by_side=%s compact=%s" % [tag, image.get_width(), image.get_height(), main.menu_layout.side_by_side, main.menu_layout.compact])
	panel.api.token = ""
	quit()
