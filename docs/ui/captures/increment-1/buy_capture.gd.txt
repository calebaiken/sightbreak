extends SceneTree
## Native captures of the requisition overlay and the live HUD. --out=<dir>
func _initialize() -> void: call_deferred("run")
func settle(n := 8) -> void:
	for i in n: await process_frame
func shot(out: String, name: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(out.path_join(name + ".png"))
	print("CAPTURE ", name)
func run() -> void:
	var out := "/tmp/buy"
	for a in OS.get_cmdline_user_args():
		if a.begins_with("--out="): out = a.trim_prefix("--out=")
	DirAccess.make_dir_recursive_absolute(out)
	var main = load("res://main.tscn").instantiate()
	root.add_child(main)
	main.profile.data.settings.touch_visible = true
	main.profile.data.settings.hud_scale = 1.0
	main.profile.data.settings.hud_positions = {}
	for config in [[Vector2i(1280, 720), 1.0], [Vector2i(960, 720), 1.0], [Vector2i(960, 720), 1.3], [Vector2i(1600, 720), 1.15]]:
		main.profile.data.settings.ui_text_scale = config[1]
		root.content_scale_size = config[0]
		root.size = config[0]
		await settle()
		main.start_match({"mode": "offline", "difficulty": "normal", "name": "Capture"})
		await settle(20)
		var tag := "%dx%d-%d" % [config[0].x, config[0].y, roundi(config[1] * 100)]
		main.match_game.local_actor().credits = 1500
		main.show_buy()
		await settle(10)
		await shot(out, tag + "-requisition")
		main.close_overlay()
		main.match_game.phase = "live"
		main.match_game.time_left = 90.0
		main.status_label.set_meta("until", 0.0)
		main.show_notification("Match complete; account reward submission failed. Server logs identify the match for review.")
		await settle(10)
		await shot(out, tag + "-hud-notice")
		main.leave_match()
		await settle()
	main.profile.data.settings.ui_text_scale = 1.0
	quit()
