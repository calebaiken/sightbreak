extends SceneTree
## Native tour of every main-shell screen for the visual/UI gap review. --out=<dir>
func _initialize() -> void: call_deferred("run")
func settle(n := 8) -> void:
	for i in n: await process_frame
func shot(out: String, name: String) -> void:
	await RenderingServer.frame_post_draw
	root.get_texture().get_image().save_png(out.path_join(name + ".png"))
	print("TOUR ", name)
func run() -> void:
	var out := "/tmp/ui-tour"
	for a in OS.get_cmdline_user_args():
		if a.begins_with("--out="): out = a.trim_prefix("--out=")
	DirAccess.make_dir_recursive_absolute(out)
	var main = load("res://main.tscn").instantiate()
	root.add_child(main)
	main.profile.data.settings.touch_visible = true
	main.profile.data.settings.ui_text_scale = 1.0
	main.profile.data.settings.hud_scale = 1.0
	main.profile.data.settings.hud_positions = {}
	await settle()
	for dims in [Vector2i(1280, 720), Vector2i(960, 720)]:
		root.content_scale_size = dims
		root.size = dims
		await settle()
		var tag := "%dx%d" % [dims.x, dims.y]
		for page in ["show_menu", "show_connect", "show_settings", "show_progress"]:
			main.call(page)
			await settle(20)
			await shot(out, "%s-%s" % [tag, page.trim_prefix("show_")])
		main.start_match({"mode": "offline", "difficulty": "normal", "name": "UI tour"})
		await settle(30)
		await shot(out, tag + "-hud-buy")
		main.show_buy()
		await settle(12)
		await shot(out, tag + "-buy-menu")
		main.close_overlay()
		main.match_game.phase = "live"
		main.match_game.time_left = 90.0
		main.status_label.set_meta("until", 0.0)
		await settle(20)
		await shot(out, tag + "-hud-live")
		main.touch_score_until = main.elapsed + 20
		main.show_scoreboard()
		await settle(8)
		await shot(out, tag + "-scoreboard")
		main.touch_score_until = 0
		if main.has_method("toggle_pause"):
			main.toggle_pause()
			await settle(8)
			await shot(out, tag + "-pause")
			main.toggle_pause()
			await settle(4)
		main.open_touch_editor()
		await settle(8)
		await shot(out, tag + "-touch-editor")
		main.finish_touch_editor()
		main.leave_match()
		await settle()
	main.queue_free()
	await settle()
	print("TOUR done")
	quit()
