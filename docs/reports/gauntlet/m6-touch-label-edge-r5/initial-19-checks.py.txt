import hashlib
import json
from pathlib import Path
import subprocess
import unittest

import numpy as np
from PIL import Image

from prototype import ACTIONS, audit_labels, neutral_difference_interval
from analyze_touch_contrast_r4b import audit_affordance, audit_boundary, audit_pixels

HERE = Path(__file__).resolve().parent
REPO = HERE.parents[3]
OLD = "108d1cdc6cf471daa7390298c234bb1c087ae865"


def read(path):
    return np.asarray(Image.open(path).convert("RGB"))


def fixture():
    shape = (80, 720, 3)
    grouped = {g: np.zeros(shape, np.uint8) for g in ("active_label", "inactive_label")}
    components, labels = {}, {}
    info = {"size": [720, 80], "counts": {"grenade": 0}, "geometry": {}}
    for i, action in enumerate(ACTIONS):
        x, y = 30 + i * 60, 40
        info["geometry"][action] = {"point": [x, y], "hit_radius": 25}
        components[action] = np.zeros(shape, np.uint8)
        labels[action] = np.zeros(shape, np.uint8)
        # Include a one-code left overhang; no nominal rectangle is consulted.
        labels[action][y, x - 2] = 1
        components[action][y, x - 2] = 255
        group = "inactive_label" if action == "grenade" else "active_label"
        grouped[group][y, x - 2] = 1
    return info, grouped, components, labels


class ProposalControls(unittest.TestCase):
    def test_keeps_every_negative_bearing_one_code_pixel(self):
        row = audit_labels(*fixture())
        self.assertTrue(row["pass"], row)
        self.assertEqual(sum(r["support_pixels"] for r in row["rows"]), 11)

    def test_dropped_low_coverage_pixel_rejects(self):
        args = fixture(); args[3]["jump"][:] = 0
        self.assertFalse(audit_labels(*args)["pass"])

    def test_duplicated_pixel_rejects_even_with_two_component_claims(self):
        info, groups, components, labels = fixture()
        labels["fire"][40, 88] = 1; components["fire"][40, 88] = 255
        self.assertFalse(audit_labels(info, groups, components, labels)["pass"])

    def test_relabeling_inactive_source_as_active_rejects(self):
        info, groups, components, labels = fixture()
        labels["jump"], labels["grenade"] = labels["grenade"], labels["jump"]
        self.assertFalse(audit_labels(info, groups, components, labels)["pass"])

    def test_foreign_component_rejects(self):
        args = fixture(); args[2]["jump"][:] = 0
        self.assertFalse(audit_labels(*args)["pass"])

    def test_extra_pixel_rejects(self):
        args = fixture(); args[3]["jump"][40, 89] = 1; args[2]["jump"][40, 89] = 255
        self.assertFalse(audit_labels(*args)["pass"])

    def test_physical_hit_spill_rejects(self):
        args = fixture(); args[0]["geometry"]["jump"]["hit_radius"] = 1
        self.assertFalse(audit_labels(*args)["pass"])

    def test_neighbor_winner_rejects_inside_original_hit(self):
        args = fixture(); args[0]["geometry"]["fire"]["point"] = [88.5, 40.5]
        self.assertFalse(audit_labels(*args)["pass"])

    def test_missing_action_non_native_shape_and_non_neutral_reject(self):
        for variant in ("missing", "shape", "neutral", "float"):
            args = fixture()
            if variant == "missing": args[3].pop("jump")
            if variant == "shape": args[3]["jump"] = np.zeros((1, 1, 3), np.uint8)
            if variant == "neutral": args[3]["jump"][40, 88] = [1, 2, 1]
            if variant == "float": args[3]["jump"] = args[3]["jump"].astype(float)
            with self.subTest(variant=variant): self.assertFalse(audit_labels(*args)["pass"])


class QuantizationControls(unittest.TestCase):
    def test_exact_neutral_and_spread_one_have_nonzero_interval(self):
        for white in ([29, 29, 29], [28, 29, 29]):
            result = neutral_difference_interval([0, 0, 0], white)
            self.assertTrue(result["feasible"])
            self.assertGreater(result["width"], 0)

    def test_spread_two_only_touches_closed_endpoints(self):
        result = neutral_difference_interval([170, 146, 94], [198, 175, 124])
        self.assertTrue(result["feasible"])
        self.assertEqual((result["lower"], result["upper"]), (29., 29.))
        self.assertTrue(result["endpoint_only"])

    def test_spread_three_rejects(self):
        self.assertFalse(neutral_difference_interval([0, 0, 0], [28, 31, 29])["feasible"])

    def test_negative_and_overflow_reject(self):
        for black, white in (([30]*3, [0]*3), ([256]*3, [0]*3)):
            if max(black) > 255:
                with self.assertRaises(ValueError): neutral_difference_interval(black, white)
            else: self.assertFalse(neutral_difference_interval(black, white)["feasible"])

    def test_final_feasibility_does_not_imply_affine_or_contrast_success(self):
        # Identical W/B gives feasible difference zero, with no measurable ink.
        self.assertTrue(neutral_difference_interval([30]*3, [30]*3)["feasible"])
        shape = (3, 3, 3)
        zero, footprint = np.zeros(shape, np.uint8), np.full(shape, 255, np.uint8)
        self.assertFalse(audit_pixels(zero, zero, zero, zero, 1., footprint)["resolved"])


class OriginalFailures(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        cls.manifest = json.loads((HERE / "retained-inputs.json").read_text())

    def test_every_preserved_input_is_exact(self):
        for entry in self.manifest["files"]:
            data = (HERE / entry["copy"]).read_bytes()
            self.assertEqual(len(data), entry["bytes"])
            self.assertEqual(hashlib.sha256(data).hexdigest(), entry["sha256"])

    def test_all_seven_original_gated_failures_remain(self):
        for row in self.manifest["gated_rows"]:
            root = HERE / "retained" / row["scene"]
            group = row["group"]
            args = [read(root / name) for name in ("normal.png", group + "-underlay.png",
                    group + "-black.png", group + "-white.png")]
            footprint = read(root / (group + "-footprint.png"))
            function = audit_boundary if group.startswith("component_") else audit_pixels
            actual = function(*args, row["primitive_alpha"], footprint=footprint)
            self.assertFalse(actual["resolved"])
            self.assertEqual(actual["nonneutral_coverage_pixels"], 1)

    def test_all_sixteen_mobile_assignment_failures_remain(self):
        for scene in self.manifest["label_scenes"]:
            root = HERE / "retained" / scene
            info = json.loads((root / "scene.json").read_text())
            footprints = {g: read(root / (g + "-footprint.png")) for g in
                ("active_label", "inactive_label", "active_icon", "inactive_icon", "empty_count", "ready_count")}
            components = {a: read(root / ("component_" + a + "-footprint.png")) for a in ACTIONS}
            actual = audit_affordance(info, footprints, components)
            self.assertFalse(actual["pass"])
            self.assertEqual(actual["errors"], ["active_label: native glyph pixels not assigned to exactly one expected source control: 3"])

    def test_proposal_cannot_replay_old_capture_without_new_native_evidence(self):
        scene = HERE / "retained" / self.manifest["label_scenes"][0]
        info = json.loads((scene / "scene.json").read_text())
        groups = {g: read(scene / (g + "-footprint.png")) for g in ("active_label", "inactive_label")}
        components = {a: read(scene / ("component_" + a + "-footprint.png")) for a in ACTIONS}
        self.assertFalse(audit_labels(info, groups, components, {})["pass"])

    def test_product_old_method_geometry_and_instrumentation_are_unchanged(self):
        for path in ("game/scripts/ui/touch_controls.gd", "scripts/analyze_touch_contrast.py",
                     "scripts/analyze_touch_contrast_r4b.py", "scripts/touch_geometry_r4b.py",
                     "scripts/prepare_touch_contrast_r4b.py", "scripts/fixtures/touch_contrast_r4b/touch_contrast_capture.gd"):
            expected = subprocess.check_output(["git", "-C", str(REPO), "show", OLD + ":" + path])
            self.assertEqual((REPO / path).read_bytes(), expected, path)


if __name__ == "__main__": unittest.main(verbosity=2)
