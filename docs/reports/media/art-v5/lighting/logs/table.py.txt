import re,sys,collections
def load(path):
    tab=collections.defaultdict(dict)
    for l in open(path):
        if not l.startswith("LIGHT"): continue
        parts=[p.strip() for p in l.split("|")]
        m=re.search(r"n=(\d+) median=([\d.]+) p10=([\d.]+) p90=([\d.]+) detail=([\d.]+)",parts[4])
        tab[(parts[1],parts[2])][parts[3]]=tuple(float(x) for x in m.groups()) if m else None
    return tab
P=["magazine","receiver","handguard","furniture","gloves","sleeves"]
def show(tab,title):
    print(title)
    print("%-14s %-11s " % ("scenario","pose")+" ".join("%-14s"%p for p in P))
    for k,d in tab.items():
        print("%-14s %-11s " % k+" ".join("%-14s" % ("-" if not d.get(p) else "%d/%d/%d" % (round(d[p][1]),round(d[p][2]),round(d[p][3]))) for p in P))
if __name__=="__main__":
    show(load(sys.argv[1]),sys.argv[1])
