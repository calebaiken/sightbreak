extends SceneTree
## Scratch: status text glyph bounds against the FRAG and SMOKE touch targets.
func _initialize() -> void: call_deferred("run")
func settle() -> void:
	for i in 6: await process_frame
func text_rect(label: Label) -> Rect2:
	# Union of the drawn lines: each line's width, centred in the label's rect.
	var font := label.get_theme_font("font")
	var size := label.get_theme_font_size("font_size")
	var rect := label.get_global_rect()
	var lines := label.get_line_count()
	var height := label.get_line_height() * lines + label.get_theme_constant("line_spacing") * maxi(0, lines - 1)
	var widest := 0.0
	for i in lines:
		var start := 0
		widest = maxf(widest, font.get_string_size(label.text, HORIZONTAL_ALIGNMENT_LEFT, -1, size).x if lines == 1 else rect.size.x)
	var top := rect.position.y + (rect.size.y - height) * 0.5
	return Rect2(rect.position.x + (rect.size.x - widest) * 0.5, top, widest, height)
func run() -> void:
	var scale := 1.3
	for a in OS.get_cmdline_user_args():
		if a.begins_with("--scale="): scale = float(a.trim_prefix("--scale="))
	var main = load("res://main.tscn").instantiate()
	root.add_child(main)
	main.profile.data.settings.touch_visible = true
	main.profile.data.settings.hud_positions = {}
	main.profile.data.settings.ui_text_scale = scale
	await settle()
	for dimensions in [Vector2i(1280, 720), Vector2i(1600, 720), Vector2i(960, 720), Vector2i(1560, 720)]:
		root.content_scale_size = dimensions
		root.size = dimensions
		await settle()
		main.start_match({"mode": "offline", "difficulty": "normal", "name": "Probe"})
		await settle()
		for phase in ["buy", "live"]:
			if phase == "live":
				main.match_game.phase = "live"
				main.status_label.set_meta("until", 0.0)
				main._process(0.01)
				await settle()
			var label: Label = main.status_label
			var glyphs := text_rect(label)
			var line := "PROBE %dx%d %s scale=%.2f lines=%d text='%s' label=%s glyphs=%s" % [dimensions.x, dimensions.y, phase, scale, label.get_line_count(), label.text, label.get_global_rect(), glyphs]
			for action in ["grenade", "smoke"]:
				var point: Vector2 = main.touch.control_point(action)
				var radius: float = main.touch.radius(action)
				var target := Rect2(point - Vector2.ONE * radius, Vector2.ONE * radius * 2)
				line += " %s=%s%s" % [action, target, " OVERLAP" if target.intersects(glyphs) else ""]
			print(line)
		main.leave_match()
		await settle()
	main.queue_free()
	await settle()
	quit()
