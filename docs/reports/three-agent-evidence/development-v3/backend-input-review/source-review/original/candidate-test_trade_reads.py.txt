15:     return str(uuid5(NAMESPACE_URL, 'sightbreak/trade-read/' + label))

18: def seed_workload(count, per_side):
19:     """Stable public payloads; no HTTP grants, rate-limit bypass or live accounts."""
20:     viewer = fixture_id('viewer')
21:     outsider = fixture_id('outsider')
22:     hidden_partner = fixture_id('hidden-partner')
23:     names = {viewer: 'Viewer', outsider: 'Outsider', hidden_partner: 'Hidden partner'}
24:     names.update({fixture_id(f'partner/{i}'): f'Partner {i}' for i in range(count)})
25:     acquired = datetime(2026, 10, 1, 12, 0, tzinfo=timezone.utc)
26:     expected = []
27:     with transaction() as db:
28:         # PostgreSQL renders timestamptz in its session timezone on every platform.
29:         acquired = db.scalar(text('SELECT CAST(:instant AS timestamptz)'), {'instant': acquired})
30:         db.add_all(Player(id=pid, username='fixture_' + str(i), display_name=name,
31:                           password_hash='fixture-unusable')
32:                    for i, (pid, name) in enumerate(names.items()))
33:         db.flush()
34:         for pid, token in ((viewer, 'trade-read-viewer'), (outsider, 'trade-read-outsider')):
35:             db.add(Session(token_hash=digest(token), player_id=pid,
36:                            expires_at=now() + timedelta(hours=1)))
37:         for i in range(count):
38:             partner = fixture_id(f'partner/{i}')
39:             a, b = (viewer, partner) if i % 2 == 0 else (partner, viewer)
40:             tid = fixture_id(f'trade/{i}')
41:             state = ('open', 'completed', 'cancelled')[i % 3]
42:             reservation = 'trade:' + tid if state == 'open' else None
43:             offers, metadata = {}, {}
44:             for owner, receiver in ((a, b), (b, a)):
45:                 ids, values = [], []
46:                 for j in range(per_side):
47:                     iid = fixture_id(f'item/{i}/{owner}/{j}')
48:                     ids.append(iid)
49:                     db.add(Item(id=iid, owner_id=receiver if state == 'completed' else owner,
50:                                 definition_id='obsidian_rare', rarity='Rare',
51:                                 pattern_seed=j, wear=1234, variant='standard',
52:                                 reservation=reservation, acquired_at=acquired))
53:                     values.append(dict(id=iid, definition_id='obsidian_rare', rarity='Rare',
54:                                        pattern_seed=j, wear=0.1234, variant='standard',
55:                                        collection='Obsidian', kind='weapon_skin',
56:                                        name='Obsidian / Carbon', reserved=reservation is not None,
57:                                        trade_eligible=reservation is None,
58:                                        marketplace_eligible=reservation is None,
59:                                        acquired_at=acquired.isoformat()))
60:                 # A batch must not replace the agreed offer order with SQL/set order.
61:                 offers[owner], metadata[owner] = ids[::-1], values[::-1]
62:             confirmations = [a] if state == 'open' else [a, b] if state == 'completed' else []
63:             db.add(Trade(id=tid, a=a, b=b, revision=i + 2, offers=offers,
64:                          confirmations=confirmations, state=state))
65:             expected.append(dict(id=tid, a=a, b=b, names={a: names[a], b: names[b]},
66:                                  revision=i + 2, offers=offers, offer_items=metadata,
67:                                  confirmations=confirmations, state=state))
68:         # Unrelated data must never enter the participant's response or name maps.
69:         db.add(Trade(id=fixture_id('hidden-trade'), a=outsider, b=hidden_partner,
70:                      offers={outsider: [], hidden_partner: []}, confirmations=[]))
71:     return {'Authorization': 'Bearer trade-read-viewer'}, expected

74: def canonical_trades(payload):
75:     # The route has never promised top-level ordering; offer arrays DO have order.
76:     return {'trades': sorted(payload['trades'], key=lambda trade: trade['id'])}

79: def database_snapshot():
80:     """All authoritative tables, not only the fields exposed by this endpoint."""
81:     with engine.connect() as db:
82:         return {name: db.execute(text(f'SELECT to_jsonb(t)::text FROM "{name}" t '
83:                                      'ORDER BY to_jsonb(t)::text')).scalars().all()
84:                 for name in sorted(Base.metadata.tables)}

87: @pytest.mark.parametrize('count,per_side', [(0, 0), (1, 0), (1, 2), (40, 1), (40, 20), (501, 0)])
88: def test_trade_read_payload_privacy_and_query_budget(client, count, per_side):
89:     headers, expected = seed_workload(count, per_side)
90:     statements = []
91:     def record(conn, cursor, statement, parameters, context, executemany):
92:         statements.append(statement)
93:     event.listen(engine, 'before_cursor_execute', record)
94:     try:
95:         response = client.get('/trades', headers=headers)
96:     finally:
97:         event.remove(engine, 'before_cursor_execute', record)
98:     assert response.status_code == 200
99:     assert canonical_trades(response.json()) == canonical_trades({'trades': expected})
100:     # Authentication is two SELECTs, then the participant-only trade query.
101:     # Additional reads scale in bounded 500-ID batches, not with each offer/name.
102:     budget = 3 + (ceil((count + 1) / 500) if count else 0) + ceil(count * per_side * 2 / 500)
103:     assert len(statements) <= budget, (len(statements), budget)
104:     assert all(statement.lstrip().upper().startswith('SELECT') for statement in statements)

107: def test_trade_reads_keep_current_metadata_for_closed_trades(client):
108:     headers, expected = seed_workload(3, 2)
109:     assert canonical_trades(client.get('/trades', headers=headers).json()) == canonical_trades({'trades': expected})
110:     closed = expected[1]
111:     partner = next(pid for pid in closed['names'] if pid != fixture_id('viewer'))
112:     offered = closed['offer_items'][closed['a']][0]
113:     consumed_key = closed['offer_items'][closed['b']][0]
114:     with transaction() as db:
115:         db.get(Player, partner).display_name = 'Updated partner'
116:         current = db.get(Item, offered['id'])
117:         current.reservation = 'market:later-listing'
118:         current.owner_id = fixture_id('outsider')
119:         key = db.get(Item, consumed_key['id'])
120:         key.definition_id = 'key'
121:         key.rarity = 'Common'
122:         key.consumed = True
123:     closed['names'][partner] = 'Updated partner'
124:     offered.update(reserved=True, trade_eligible=False, marketplace_eligible=False)
125:     consumed_key.update(definition_id='key', rarity='Common', collection='Universal',
126:                         kind='key', name='Collection key', trade_eligible=False,
127:                         marketplace_eligible=False)
128:     assert canonical_trades(client.get('/trades', headers=headers).json()) == canonical_trades({'trades': expected})

131: def test_trade_reads_preserve_repeated_historical_items(client):
132:     headers, expected = seed_workload(3, 2)
133:     viewer = fixture_id('viewer')
134:     # A once-traded item can appear in several historical offers. It is serialized
135:     # from the current Item record in each original position, just as before.
136:     source, target = expected[1], expected[2]
137:     target['offers'][viewer] = source['offers'][viewer][::-1]
138:     target['offer_items'][viewer] = source['offer_items'][viewer][::-1]
139:     with transaction() as db:
140:         db.get(Trade, target['id']).offers = target['offers']
141:     assert canonical_trades(client.get('/trades', headers=headers).json()) == canonical_trades({'trades': expected})

156: @pytest.mark.parametrize('condition,status', [('missing', 401), ('expired', 401), ('revoked', 401), ('inactive', 403)])
157: def test_trade_read_rechecks_authentication(client, condition, status):
158:     headers, _ = seed_workload(1, 1)
159:     with transaction() as db:
160:         if condition == 'missing':
161:             headers = {}
162:         elif condition == 'expired':
163:             db.get(Session, digest('trade-read-viewer')).expires_at = now() - timedelta(seconds=1)
164:         elif condition == 'revoked':
165:             db.delete(db.get(Session, digest('trade-read-viewer')))
166:         else:
167:             db.get(Player, fixture_id('viewer')).active = False
168:     assert client.get('/trades', headers=headers).status_code == status
